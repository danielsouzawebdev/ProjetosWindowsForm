Projeto 2 · Calculadora de IMC

Objetivo: proteger os dados da classe com private e validar valores.

Tela

txtPeso, txtAltura, btnCalcular, lblIMC, lblClassificacao.

Classe Paciente

public class Paciente
{
    // CAMPOS PRIVADOS: só a própria classe enxerga. Isso é ENCAPSULAMENTO.
    private double peso;
    private double altura;

    // A propriedade é a "porta de entrada" controlada para o campo.
    public double Peso
    {
        get { return peso; }
        set
        {
            // "value" é o valor que alguém tentou atribuir.
            if (value <= 0)
                throw new ArgumentException("Peso inválido"); // interrompe e avisa o erro
            peso = value;
        }
    }

    public double Altura
    {
        get { return altura; }
        set
        {
            if (value <= 0)
                throw new ArgumentException("Altura inválida");
            altura = value;
        }
    }

    // "=>" é uma forma curta de escrever um método de uma linha.
    public double CalcularIMC() => peso / (altura * altura);

    public string Classificar()
    {
        double imc = CalcularIMC();
        if (imc < 18.5) return "Abaixo do peso";
        if (imc < 25)   return "Peso normal";
        if (imc < 30)   return "Sobrepeso";
        return "Obesidade";
    }
}

Botão com tratamento de erros

private void btnCalcular_Click(object sender, EventArgs e)
{
    // TryParse tenta converter SEM quebrar. Retorna false se não for número.
    // "out double peso" recebe o valor convertido.
    if (!double.TryParse(txtPeso.Text, out double peso) ||
        !double.TryParse(txtAltura.Text, out double altura))
    {
        MessageBox.Show("Digite apenas números.");
        return; // sai do método
    }

    try // tenta executar...
    {
        // Inicializador de objeto: define as propriedades na criação.
        Paciente p = new Paciente { Peso = peso, Altura = altura };
        lblIMC.Text = p.CalcularIMC().ToString("F2"); // F2 = 2 casas decimais
        lblClassificacao.Text = p.Classificar();
    }
    catch (ArgumentException ex) // ...e se a classe lançar erro, cai aqui
    {
        MessageBox.Show(ex.Message);
    }
}
💡 Com private, ninguém de fora coloca um valor inválido: a classe protege os próprios dados.
