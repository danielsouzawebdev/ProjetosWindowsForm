Objetivo: juntar tudo: classes que contêm outras, cálculos e validação.

Tela

txtAluno, btnCriarAluno, txtNota, btnAddNota, lstNotas, lblMedia, lblSituacao.

Classe Aluno

using System.Linq;

public class Aluno
{
    public string Nome { get; set; }

    // COMPOSIÇÃO: o aluno "tem" uma lista de notas.
    // IReadOnlyList impede que de fora alterem a lista sem passar pelas regras.
    private List<double> notas = new List<double>();
    public IReadOnlyList<double> Notas => notas;

    public void AdicionarNota(double nota)
    {
        if (nota < 0 || nota > 10)
            throw new ArgumentException("A nota deve estar entre 0 e 10.");
        notas.Add(nota);
    }

    // Propriedade calculada: sempre reflete as notas atuais.
    public double Media => notas.Count == 0 ? 0 : notas.Average();

    public string Situacao
    {
        get
        {
            if (notas.Count == 0)  return "Sem notas";
            if (Media >= 7)        return "Aprovado";
            if (Media >= 5)        return "Recuperação";
            return "Reprovado";
        }
    }
}

Formulário

Aluno aluno;   // ainda null: nenhum aluno criado

private void btnCriarAluno_Click(object sender, EventArgs e)
{
    aluno = new Aluno { Nome = txtAluno.Text };
    lstNotas.Items.Clear();
    AtualizarResumo();
}

private void btnAddNota_Click(object sender, EventArgs e)
{
    if (aluno == null)   // evita erro de referência nula
    {
        MessageBox.Show("Crie o aluno primeiro.");
        return;
    }
    try
    {
        double nota = double.Parse(txtNota.Text);
        aluno.AdicionarNota(nota);
        lstNotas.Items.Add(nota);
        AtualizarResumo();
    }
    catch (Exception ex)   // pega erro de formato E a regra da classe
    {
        MessageBox.Show(ex.Message);
    }
}

private void AtualizarResumo()
{
    lblMedia.Text = $"Média: {aluno.Media:F1}";
    lblSituacao.Text = aluno.Situacao;
}
