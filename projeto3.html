Projeto 3 · Conversor de Temperatura

Objetivo: usar uma classe utilitária com métodos static.

Tela

txtValor, cmbModo (Items: C para F, C para K, F para C), btnConverter, lblResultado.

Classe estática

// static class: não se cria objeto com new, usamos direto Conversor.Metodo().
// Ideal para funções utilitárias que não guardam dados.
public static class Conversor
{
    public static double CelsiusParaFahrenheit(double c) => c * 9 / 5 + 32;
    public static double CelsiusParaKelvin(double c)     => c + 273.15;
    public static double FahrenheitParaCelsius(double f) => (f - 32) * 5 / 9;
}

Botão

private void btnConverter_Click(object sender, EventArgs e)
{
    if (!double.TryParse(txtValor.Text, out double valor))
    {
        MessageBox.Show("Valor inválido.");
        return;
    }

    double resultado;

    // switch escolhe o caminho conforme o texto selecionado no ComboBox.
    switch (cmbModo.Text)
    {
        case "C para F": resultado = Conversor.CelsiusParaFahrenheit(valor); break;
        case "C para K": resultado = Conversor.CelsiusParaKelvin(valor);     break;
        case "F para C": resultado = Conversor.FahrenheitParaCelsius(valor); break;
        default:
            MessageBox.Show("Escolha uma conversão.");
            return;
    }
    lblResultado.Text = resultado.ToString("F1");
}
💡 Métodos static pertencem à classe, não ao objeto. Math.Sqrt() é um exemplo que você já usa.
