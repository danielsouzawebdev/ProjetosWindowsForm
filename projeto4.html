Projeto 4 · Lista de Tarefas

Objetivo: guardar vários objetos em uma coleção e exibi-los.

Tela

txtTarefa, btnAdicionar, lstTarefas (ListBox), btnConcluir, btnRemover.

Classe Tarefa

public class Tarefa
{
    public string Descricao { get; set; }
    public bool Concluida { get; set; }   // começa como false

    // override: troca o comportamento herdado de object.
    // O ListBox usa o ToString() para decidir o que mostrar na lista.
    public override string ToString()
    {
        return (Concluida ? "[X] " : "[  ] ") + Descricao; // ? : é um if curto
    }
}

Formulário

// List<Tarefa> é uma lista que só aceita objetos do tipo Tarefa.
// Declarada fora dos métodos para existir enquanto o form estiver aberto.
List<Tarefa> tarefas = new List<Tarefa>();

private void btnAdicionar_Click(object sender, EventArgs e)
{
    if (string.IsNullOrWhiteSpace(txtTarefa.Text)) return; // ignora vazio

    tarefas.Add(new Tarefa { Descricao = txtTarefa.Text }); // adiciona na lista
    txtTarefa.Clear();
    AtualizarLista();
}

private void btnConcluir_Click(object sender, EventArgs e)
{
    // SelectedItem devolve object; "as Tarefa" converte de volta ao tipo certo.
    Tarefa t = lstTarefas.SelectedItem as Tarefa;
    if (t == null) return;       // nada selecionado
    t.Concluida = true;
    AtualizarLista();
}

private void btnRemover_Click(object sender, EventArgs e)
{
    Tarefa t = lstTarefas.SelectedItem as Tarefa;
    if (t != null) tarefas.Remove(t);
    AtualizarLista();
}

// Método auxiliar: reconstrói o ListBox a partir da lista.
private void AtualizarLista()
{
    lstTarefas.Items.Clear();
    foreach (Tarefa t in tarefas)      // foreach percorre item por item
        lstTarefas.Items.Add(t);
}
💡 Separe dados (a lista) de exibição (o ListBox) e use um método como AtualizarLista() para sincronizar.
