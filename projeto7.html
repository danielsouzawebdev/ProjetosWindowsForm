Projeto 7 · Formas Geométricas

Objetivo: definir um contrato comum com abstract.

Tela

cmbForma (Círculo, Retângulo), txtA, txtB, btnCalcular, lstFormas.

Classes

// abstract: não pode ser instanciada (new Forma() dá erro).
// Serve apenas de modelo para as filhas.
public abstract class Forma
{
    public string Nome { get; protected set; }

    // abstract: sem corpo. TODA filha é OBRIGADA a implementar.
    public abstract double Area();
    public abstract double Perimetro();

    public override string ToString() =>
        $"{Nome}: área {Area():F2} | perímetro {Perimetro():F2}";
}

public class Circulo : Forma
{
    private double raio;
    public Circulo(double raio) { this.raio = raio; Nome = "Círculo"; }

    public override double Area()      => Math.PI * raio * raio;
    public override double Perimetro() => 2 * Math.PI * raio;
}

public class Retangulo : Forma
{
    private double largura, altura;
    public Retangulo(double l, double a) { largura = l; altura = a; Nome = "Retângulo"; }

    public override double Area()      => largura * altura;
    public override double Perimetro() => 2 * (largura + altura);
}

Botão

private void btnCalcular_Click(object sender, EventArgs e)
{
    double a = double.Parse(txtA.Text);
    double b = string.IsNullOrEmpty(txtB.Text) ? 0 : double.Parse(txtB.Text);

    // A variável é do tipo Forma, mas guarda um Circulo ou Retangulo.
    Forma f = cmbForma.Text == "Círculo" ? new Circulo(a) : new Retangulo(a, b);

    lstFormas.Items.Add(f); // exibe usando o ToString() de cada forma
}
💡 Classe abstrata = contrato obrigatório. Quem usa Forma não precisa saber qual é a forma real.
