Projeto 5 · Controle de Estoque

Objetivo: fazer a classe proteger a regra: estoque nunca fica negativo.

Tela

txtNome, txtQtd, btnNovo, btnEntrada, btnSaida, dgvProdutos (DataGridView).

Classe Produto

public class Produto
{
    public string Nome { get; set; }

    // "private set": qualquer um lê a quantidade, mas só a classe altera.
    public int Quantidade { get; private set; }

    // Propriedade calculada: sem campo, só get.
    public string Situacao => Quantidade < 5 ? "ESTOQUE BAIXO" : "OK";

    public void Entrada(int qtd)
    {
        if (qtd > 0) Quantidade += qtd;
    }

    // Retorna bool para informar se a operação deu certo.
    public bool Saida(int qtd)
    {
        if (qtd <= 0 || qtd > Quantidade) return false; // regra de negócio
        Quantidade -= qtd;
        return true;
    }
}

Formulário

// BindingList atualiza o DataGridView sozinho quando a lista muda.
BindingList<Produto> produtos = new BindingList<Produto>();

private void Form1_Load(object sender, EventArgs e)
{
    dgvProdutos.DataSource = produtos;   // liga a grade à lista
}

private void btnNovo_Click(object sender, EventArgs e)
{
    produtos.Add(new Produto { Nome = txtNome.Text });
}

private void btnSaida_Click(object sender, EventArgs e)
{
    Produto p = (Produto)dgvProdutos.CurrentRow.DataBoundItem; // objeto da linha
    if (!p.Saida(int.Parse(txtQtd.Text)))
        MessageBox.Show("Quantidade indisponível!");
    dgvProdutos.Refresh();   // redesenha para mostrar o novo valor
}
💡 Regras importantes ficam na classe, não no botão. Assim qualquer tela que use Produto respeita a regra. (Precisa de using System.ComponentModel;.)
