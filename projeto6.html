Projeto 6 · Contas Bancárias

Objetivo: criar contas diferentes reaproveitando uma classe base.

Tela

txtTitular, txtValor, cmbTipo (Corrente, Poupança), btnCriar, btnDepositar, btnSacar, dgvContas.

Classe base Conta

public class Conta
{
    public string Titular { get; set; }

    // protected set: a classe e suas FILHAS podem alterar o saldo.
    public decimal Saldo { get; protected set; }

    public void Depositar(decimal valor)
    {
        if (valor > 0) Saldo += valor;
    }

    // virtual: permite que as classes filhas REESCREVAM este método.
    public virtual bool Sacar(decimal valor)
    {
        if (valor <= 0 || valor > Saldo) return false;
        Saldo -= valor;
        return true;
    }
}

Classes filhas

// ": Conta" significa HERANÇA: ContaCorrente já tem Titular, Saldo, Depositar...
public class ContaCorrente : Conta
{
    public decimal Limite { get; set; } = 500;

    // override: mesma assinatura, comportamento próprio (cheque especial).
    public override bool Sacar(decimal valor)
    {
        if (valor <= 0 || valor > Saldo + Limite) return false;
        Saldo -= valor;
        return true;
    }
}

public class ContaPoupanca : Conta
{
    // Método exclusivo da poupança.
    public void Rendimento(decimal taxa) => Saldo += Saldo * taxa;
}

Formulário

// A lista é de Conta, mas aceita QUALQUER filha de Conta.
BindingList<Conta> contas = new BindingList<Conta>();

private void btnCriar_Click(object sender, EventArgs e)
{
    Conta c;
    if (cmbTipo.Text == "Corrente") c = new ContaCorrente();
    else                            c = new ContaPoupanca();
    c.Titular = txtTitular.Text;
    contas.Add(c);
}

private void btnSacar_Click(object sender, EventArgs e)
{
    Conta c = (Conta)dgvContas.CurrentRow.DataBoundItem;

    // POLIMORFISMO: chamamos Sacar() sem saber o tipo exato;
    // o C# executa a versão correta (corrente ou poupança).
    if (!c.Sacar(decimal.Parse(txtValor.Text)))
        MessageBox.Show("Saque não permitido.");
    dgvContas.Refresh();
}
💡 Em Form1_Load faça dgvContas.DataSource = contas; como no projeto anterior.
