Projeto 9 · Jogo de Adivinhação

Objetivo: modelar um jogo cuja classe lembra o próprio estado.

Tela

txtPalpite, btnTentar, btnNovoJogo, lblDica, lblTentativas.

Classe Jogo

// enum: conjunto fixo de valores com nome. Mais legível que 0, 1, 2.
public enum Resultado { Acertou, MuitoBaixo, MuitoAlto }

public class Jogo
{
    private int numeroSecreto;
    private static Random sorteio = new Random(); // um só Random para todos

    public int Tentativas { get; private set; }   // o objeto lembra seu estado

    public Jogo()
    {
        numeroSecreto = sorteio.Next(1, 101);     // sorteia de 1 a 100
    }

    public Resultado Tentar(int palpite)
    {
        Tentativas++;                              // conta mais uma tentativa
        if (palpite == numeroSecreto) return Resultado.Acertou;
        if (palpite < numeroSecreto)  return Resultado.MuitoBaixo;
        return Resultado.MuitoAlto;
    }
}

Formulário

Jogo jogo = new Jogo();    // já começa com um jogo novo

private void btnTentar_Click(object sender, EventArgs e)
{
    if (!int.TryParse(txtPalpite.Text, out int palpite)) return;

    Resultado r = jogo.Tentar(palpite);

    switch (r)
    {
        case Resultado.Acertou:    lblDica.Text = "🎉 Acertou!"; break;
        case Resultado.MuitoBaixo: lblDica.Text = "Tente um número MAIOR"; break;
        case Resultado.MuitoAlto:  lblDica.Text = "Tente um número MENOR"; break;
    }
    lblTentativas.Text = "Tentativas: " + jogo.Tentativas;
}

private void btnNovoJogo_Click(object sender, EventArgs e)
{
    jogo = new Jogo();     // novo objeto = estado zerado
    lblDica.Text = "";
    lblTentativas.Text = "Tentativas: 0";
}
💡 Cada objeto tem seu estado. Criar outro Jogo reinicia tudo sem precisar “limpar” variáveis.
🏆 Desafio: Limite a 7 tentativas: se estourar, mostre o número secreto (crie a propriedade necessária).
